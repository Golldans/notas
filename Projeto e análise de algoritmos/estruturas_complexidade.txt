b) Inserir n elementos em um vetor (por exemplo, lendo um a um a partir de uma entrada padrão).

inserirNElementosANaoSerMenosUm(vetor, tamanhoVetor) {
  contadorTamanhoVetor = 0;

  while (vetor[contadorTamanhoVetor] != null) {
    contadorTamanhoVetor++;
  }

  inputVetor = 0;
  imprime("Digite o valor que quer inserir no vetor, -1 para sair");
  leia(inputVetor);

  while(inputVetor != -1 && contadorTamanhoVetor < tamanhoVetor) {
    vetor[contadorTamanhoVetor] = inputVetor;
    imprime("Insira o próximo valor");
    leia(inputVetor);
    contadorTamanhoVetor++;
  }

  imprime("Insercao do vetor concluida!")
}

c) Inserir o elemento X em um vetor ordenado, mantendo a ordenação. Isto é eficiente?

insereNoVetorOrdenado(vetor, tamanhoVetor, elemento) {
  posicaoAtualVetor = 0;

  while(vetor[posicaoAtualVetor] <= elemento) {
    posicaoAtualVetor++;
  }

  valorAuxiliar = elemento;

  while (valorAuxiliar != null) {
    valorASerTrocado = vetor[posicaoAtualVetor];
    vetor[posicaoAtualVetor] = valorAuxiliar;
    valorAuxiliar = valorASerTrocado;
    posicaoAtualVetor++;
  }
}
